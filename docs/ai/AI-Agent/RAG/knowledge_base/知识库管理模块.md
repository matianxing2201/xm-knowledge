# 知识库管理模块

负责知识库的增删改查，文本分割、向量化、存 Milvus 这些事。

> 📁 [项目源码](https://github.com/matianxing2201/agent_practice/tree/main/app/blueprints/rag/knowledge_base)

## 目录结构

```
knowledge_base/
├── controllers.py      # 路由，6个接口
├── services.py         # 业务逻辑
└── milvus_store.py     # Milvus 读写
```

## Milvus Schema

```python
# milvus_store.py — ensure_collection
schema = client.create_schema(auto_id=False, enable_dynamic_field=False)
schema.add_field("id", DataType.INT64, is_primary=True)           # 应用自己分配id
schema.add_field("vector", DataType.FLOAT_VECTOR, dim=2048)       # 2048维向量
schema.add_field("text", DataType.VARCHAR, max_length=65535)      # 原文
```

没用 Milvus 的 auto_id，因为 auto_id 的话 upsert 会重新生成 id，Update 时 id 就变了。应用自己分配 id 就没这个问题。

## 文本分割

```python
# services.py — split_cases
_CASE_START = re.compile(r"^\d+\s")  # 匹配 "1 感冒・风寒束表证" 这种编号行

def split_cases(content: str) -> list[str]:
    cases: list[str] = []
    current: list[str] = []
    for line in content.strip().splitlines():
        if _CASE_START.match(line):
            if current:
                cases.append("\n".join(current))
            current = [line]
        else:
            current.append(line)
    if current:
        cases.append("\n".join(current))
    return cases
```

逻辑就是遇到编号行就开始新病例，其他行都追加到当前病例里。

## 向量化

```python
# services.py — embed_text
def embed_text(text: str) -> list[float]:
    from openai import OpenAI
    client = OpenAI(
        api_key=current_app.config["EMBEDDING_API_KEY"],
        base_url=current_app.config["EMBEDDING_BASE_URL"],
    )
    resp = client.embeddings.create(
        model=current_app.config["EMBEDDING_MODEL"],
        input=text,
        dimensions=current_app.config["EMBEDDING_DIM"],
    )
    return resp.data[0].embedding
```

调智谱的 embedding-3 接口，测试的时候可以替换这个函数。

## 批量导入

```python
# services.py — import_file
def import_file(content: str) -> int:
    cases = split_cases(content)
    vectors = [embed_text(case) for case in cases]
    _store().insert(cases, vectors)
    return len(cases)
```

先分割，再逐条向量化，最后批量塞进 Milvus。

## CRUD

```python
# services.py
def create_record(text: str) -> int:
    vector = embed_text(text)
    ids = _store().insert([text], [vector])
    return ids[0]

def get_record(record_id: int) -> dict | None:
    return _store().get(record_id)

def update_record(record_id: int, text: str) -> bool:
    store = _store()
    if store.get(record_id) is None:
        return False
    vector = embed_text(text)
    return store.update(record_id, text, vector)

def delete_record(record_id: int) -> bool:
    return _store().delete(record_id)

def list_records(page: int, size: int) -> dict:
    items, total = _store().list_records(page, size)
    return {"items": items, "total": total, "page": page, "size": size}
```

更新的时候会重新向量化，id 不变。

## Milvus 读写

### 分配 id

```python
# milvus_store.py — _next_ids
def _next_ids(self, n: int) -> list[int]:
    self.ensure_collection()
    rows = self.client.query(
        self.collection_name,
        filter="id >= 0",
        output_fields=["id"],
        limit=16384,
        consistency_level="Strong",
    )
    current_max = max(r["id"] for r in rows) if rows else 0
    return list(range(current_max + 1, current_max + 1 + n))
```

数据量小，直接拉全部 id 找最大值，然后往后排。

### 写入

```python
# milvus_store.py — insert
def insert(self, texts: list[str], vectors: list[list[float]]) -> list[int]:
    self.ensure_collection()
    ids = self._next_ids(len(texts))
    rows = [
        {"id": i, "vector": v, "text": t}
        for i, t, v in zip(ids, texts, vectors)
    ]
    self.client.insert(self.collection_name, data=rows)
    self.client.flush(self.collection_name)  # 不 flush 的话查不到
    return ids
```

**记得 flush**，不然写完查不到。

### 查询

```python
# milvus_store.py — get
def get(self, record_id: int) -> dict | None:
    self.ensure_collection()
    rows = self.client.query(
        self.collection_name,
        filter=f"id == {record_id}",
        output_fields=["id", "text"],
        consistency_level="Strong",
    )
    return rows[0] if rows else None
```

### 更新

```python
# milvus_store.py — update
def update(self, record_id: int, text: str, vector: list[float]) -> bool:
    if self.get(record_id) is None:
        return False
    self.client.upsert(
        self.collection_name,
        data=[{"id": record_id, "vector": vector, "text": text}],
    )
    self.client.flush(self.collection_name)
    return True
```

用 upsert，id 不变，只更新内容。

### 分页

```python
# milvus_store.py — list_records
def list_records(self, page: int, size: int) -> tuple[list[dict], int]:
    self.ensure_collection()
    offset = (page - 1) * size
    items = self.client.query(
        self.collection_name,
        filter="id >= 0",
        output_fields=["id", "text"],
        limit=size,
        offset=offset,
        consistency_level="Strong",
    )
    total = self.client.query(
        self.collection_name,
        filter="id >= 0",
        output_fields=["count(*)"],
        consistency_level="Strong",
    )[0]["count(*)"]
    return items, total
```

## 接口

6个接口，标准 CRUD：

| 方法   | 路径                          | 功能           |
| ------ | ----------------------------- | -------------- |
| POST   | /rag/knowledge/upload         | .txt 文件批量导入 |
| POST   | /rag/knowledge                | 手动输入单条   |
| GET    | /rag/knowledge                | 分页列表       |
| GET    | /rag/knowledge/\<id\>           | 单条详情       |
| PUT    | /rag/knowledge/\<id\>           | 改 text（自动重新向量化） |
| DELETE | /rag/knowledge/\<id\>           | 按 id 删除     |

路由就是取参数、校验、调 service、返回，没别的东西。

## 踩过的坑

### Milvus 写完查不到

```python
client.upsert(...)           # 写入
client.search(...)           # 结果为空！！
client.flush(...)            # 加上这行才行
```

Milvus 写入是异步的，不 flush 就查不到。

### auto_id 不能 upsert

```python
# 错误
schema.add_field("id", DataType.INT64, is_primary=True, auto_id=True)
# upsert 会重新生成 id，更新后 id 变了

# 正确
schema.add_field("id", DataType.INT64, is_primary=True)
# 应用自己分配 id
```

### consistency_level

```python
# 强一致性：写完立马能查到
self.client.query(..., consistency_level="Strong")

# 最终一致性：可能读到旧数据，但快
self.client.query(..., consistency_level="Eventually")
```

数据量小就用 Strong，高并发再考虑 Eventually。
